\begin{tikzpicture}
\begin{groupplot}[group style={group size=2 by 1, horizontal sep=6mm}, paper, height=58mm, ytick={1,...,6},

\nextgroupplot[width=88mm, xbar stacked, /pgf/bar width=3.6mm, xmin=0, xmax=3600,
    yticklabels from table={generated/vim-records.csv}{arm}, xlabel={training records},
    legend style={at={(0.5,1.03)}, anchor=south}, legend columns=2]
  \addplot[draw=black!70, fill=okOrange!80] table[x=teacher, y=y] {generated/vim-records.csv};
  \addlegendentry{teacher (Qwen3.8-27B-FP8)}
  \addplot[draw=black!70, fill=okBlue!70] table[x=student, y=y] {generated/vim-records.csv};
  \addlegendentry{student (Qwen3-4B)}
\nextgroupplot[width=52mm, xmin=0, xmax=5.5, yticklabels={}, xlabel={exam attempts passed (\%)},

  \addplot[only marks, mark=diamond*, mark size=2.6pt, okVermillion, error bars/.cd, x dir=both, x explicit]
    table[y=y, x=exam_pct, x error plus expr=\thisrow{hi_pct}-\thisrow{exam_pct},
