% Figure: generation 1 of the Vim family per arm and seed, on the sealed exams and on the held-out authored tasks, each
% against its paired base runs. Data: generated/vim-transfer-{exam,t1}-seeds.csv and vim-transfer-pooled.csv.
\begin{tikzpicture}
\pgfplotsset{arms/.style={paper, width=72mm, height=64mm, ymin=0.35, ymax=5.85, ytick={1,2,3,4,5}, ymajorgrids=false,
    xmajorgrids, xtick pos=bottom, clip=false}}
% arms top to bottom in the order of the tables: y = 6 - x
\begin{groupplot}[group style={group size=2 by 1, horizontal sep=8mm}]
\nextgroupplot[arms, title={100 sealed exams}, xlabel={exam attempts passed (\%)}, xmin=0, xmax=7.5,
    yticklabels={outcome first, shuffled requests, productive actions, one per task, all verified},
    legend to name=armslegend, legend columns=3,
    legend style={/tikz/every even column/.append style={column sep=4mm}}]
  \addplot[only marks, mark=*, mark size=1.5pt, okBlue, discard if not={side}{candidate},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-exam-seeds.csv};
  \addlegendentry{trained, per seed}
  \addplot[only marks, mark=square, mark size=1.5pt, okGrey, discard if not={side}{baseline},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-exam-seeds.csv};
  \addlegendentry{paired base, per seed}
  \addplot[only marks, mark=diamond*, mark size=2.8pt, okVermillion, discard if not={eval}{exam},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}-0.42, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-pooled.csv};
  \addlegendentry{3 seeds pooled}
\nextgroupplot[arms, title={held-out authored tasks}, xlabel={attempts passed (\%)}, xmin=0, xmax=80, yticklabels={}]
  \addplot[only marks, mark=*, mark size=1.5pt, okBlue, discard if not={side}{candidate},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-t1-seeds.csv};
  \addplot[only marks, mark=square, mark size=1.5pt, okGrey, discard if not={side}{baseline},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-t1-seeds.csv};
  \addplot[only marks, mark=diamond*, mark size=2.8pt, okVermillion, discard if not={eval}{t1},
    error bars/.cd, x dir=both, x explicit]
    table[y expr=6-\thisrow{x}-0.42, x expr=100*\thisrow{rate}, x error plus expr=100*(\thisrow{hi}-\thisrow{rate}),
      x error minus expr=100*(\thisrow{rate}-\thisrow{lo})] {generated/vim-transfer-pooled.csv};
\end{groupplot}
